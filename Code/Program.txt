        ############################################
        # Setup: arr[0..7] = {7,3,9,1,4,8,2,6}
        # DM is WORD-ADDRESSED; PC uses PC+4 branches
        ############################################

        # arr[0] = 7
        ADDI  $t0, $zero, 7
        SW    $t0, 0($zero)

        # arr[1] = 3
        ADDI  $t0, $zero, 3
        SW    $t0, 1($zero)

        # arr[2] = 9
        ADDI  $t0, $zero, 9
        SW    $t0, 2($zero)

        # arr[3] = 1
        ADDI  $t0, $zero, 1
        SW    $t0, 3($zero)

        # arr[4] = 4
        ADDI  $t0, $zero, 4
        SW    $t0, 4($zero)

        # arr[5] = 8
        ADDI  $t0, $zero, 8
        SW    $t0, 5($zero)

        # arr[6] = 2
        ADDI  $t0, $zero, 2
        SW    $t0, 6($zero)

        # arr[7] = 6
        ADDI  $t0, $zero, 6
        SW    $t0, 7($zero)

        ############################################
        # Sum arr[0..7] -> DM[8]  (sum in $t1)
        ############################################
        ADDI  $t1, $zero, 0      # sum = 0
        ADDI  $t3, $zero, 0      # i = 0
        ADDI  $t5, $zero, 8      # count = 8

sum_loop:
        LW    $t4, 0($t3)        # t4 = arr[i]
        ADD   $t1, $t1, $t4      # sum += t4
        ADDI  $t3, $t3, 1        # i++
        ADDI  $t5, $t5, -1       # count--
        ADDI  $zero, $zero, 0    # NOP (RF timing spacer)
        BEQ   $t5, $zero, sum_done
        BEQ   $zero, $zero, sum_loop

sum_done:
        SW    $t1, 8($zero)      # DM[8] = sum  (expect 40)

        ############################################
        # Reverse arr in place with 4 swaps
        # i in $t3, j in $t5, loop count in $t6
        ############################################
        ADDI  $t3, $zero, 0      # i = 0
        ADDI  $t5, $zero, 7      # j = 7
        ADDI  $t6, $zero, 4      # c = 4 swaps

rev_loop:
        LW    $t1, 0($t3)        # t1 = arr[i]
        LW    $t2, 0($t5)        # t2 = arr[j]
        SW    $t2, 0($t3)        # arr[i] = t2
        SW    $t1, 0($t5)        # arr[j] = t1
        ADDI  $t3, $t3, 1        # i++
        ADDI  $t5, $t5, -1       # j--
        ADDI  $t6, $t6, -1       # c--
        ADDI  $zero, $zero, 0    # NOP (RF timing spacer)
        BEQ   $t6, $zero, sum2_init
        BEQ   $zero, $zero, rev_loop

        ############################################
        # Sum again after reverse -> DM[9]
        ############################################
sum2_init:
        ADDI  $t1, $zero, 0      # sum = 0
        ADDI  $t3, $zero, 0      # i = 0
        ADDI  $t5, $zero, 8      # count = 8

sum2_loop:
        LW    $t4, 0($t3)        # t4 = arr[i]
        ADD   $t1, $t1, $t4      # sum += t4
        ADDI  $t3, $t3, 1        # i++
        ADDI  $t5, $t5, -1       # count--
        ADDI  $zero, $zero, 0    # NOP
        BEQ   $t5, $zero, sum2_done
        BEQ   $zero, $zero, sum2_loop

sum2_done:
        SW    $t1, 9($zero)      # DM[9] = sum (expect 40)

        .word 0xFFFFFFFF
